1 Route Setup (web.php)

Tambahkan route ini agar halaman laporan bisa diakses. 

use App\Http\Controllers\LaporanController;

// Pastikan dilindungi auth jika aplikasi sudah ada login
Route::middleware(['auth'])->group(function () {
    Route::get('/laporan', [LaporanController::class, 'index'])->name('laporan.index');
});

2 Logic Laporan (Controller)

File: app/Http/Controllers/LaporanController.php
Buat controller baru: php artisan make:controller LaporanController. 

public function index(Request $request)
{
    // 1. Siapkan Query Builder (Belum dieksekusi)
    $query = \App\Models\Product::query();

    // 2. Filter Dinamis: Jika user memilih kategori
    if ($request->filled('category_id')) {
        $query->where('category_id', $request->category_id);
    }

    // 3. Filter Stok: Jika user input angka maksimal stok
    if ($request->filled('stock_max')) {
        $query->where('stock', '<=', $request->stock_max);
    }

    // 4. Eksekusi Query (Get Data)
    $products = $query->get();

    // 5. Hitung Ringkasan (Aggregates Manual)
    // Total Nilai Aset = Sum (Harga * Stok) tiap produk
    $totalAset = $products->sum(function($product) {
        return $product->price * $product->stock;
    });

    $totalItem = $products->sum('stock');

    // 6. Ambil data kategori untuk dropdown filter
    $categories = \App\Models\Category::all();

    return view('laporan.index', compact('products', 'categories', 'totalAset', 'totalItem'));
}
