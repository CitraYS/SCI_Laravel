3 Routes (Jalur Akses)

Tambahkan di routes/web.php. Perhatikan urutan agar tidak error.

use App\Http\Controllers\ProdukController;

// 1. Tampilkan Tabel & Detail
Route::get('/produk', [ProdukController::class, 'index']);
Route::get('/produk/{id}', [ProdukController::class, 'show'])->where('id', '[0-9]+'); 

// 2. Tambah Data (Create & Store)
Route::get('/produk/create', [ProdukController::class, 'create']);
Route::post('/produk', [ProdukController::class, 'store']);

// 3. Edit Data (Edit & Update)
Route::get('/produk/{id}/edit', [ProdukController::class, 'edit']);
Route::put('/produk/{id}', [ProdukController::class, 'update']);

// 4. Hapus Data (Destroy)
Route::delete('/produk/{id}', [ProdukController::class, 'destroy']);

4 Controller Logic

// --- CREATE (Tambah) ---

public function create() {
    return view('produk.create');
}

public function store(Request $request) {
    $request->validate([
        'nama' => 'required',
        'harga' => 'required|numeric',
        'stok' => 'required|integer'
    ]);

    \App\Models\Produk::create($request->all());
    return redirect('/produk')->with('success', 'Data Berhasil Disimpan');
}

// --- EDIT (Ubah) ---

public function edit($id) {
    $produk = \App\Models\Produk::find($id);
    return view('produk.edit', compact('produk'));
}

public function update(Request $request, $id) {
    $produk = \App\Models\Produk::find($id);
    $produk->update($request->all());
    return redirect('/produk')->with('success', 'Data Berhasil Diupdate');
}

// --- DELETE (Hapus) ---

public function destroy($id) {
    $produk = \App\Models\Produk::find($id);
    $produk->delete();
    return redirect('/produk')->with('success', 'Data Berhasil Dihapus');
}
